###############################################################################
#  OSTEOPERU API - Colección de endpoints (curl)
#  BASE: http://localhost:7090/back-envio-resultados-api
#  Auth: JWT Bearer. Primero LOGIN, copia el accessToken en TOKEN.
#  Roles: ADMIN | RESULTADOS | ADMISION | PACIENTE | CALLCENTER
#         (ADMIN accede a TODO)
###############################################################################

BASE="http://localhost:7090/back-envio-resultados-api"
TOKEN="pega-aqui-el-accessToken-del-login"


# =============================================================================
# AUTENTICACION  (/api/auth)   ->   PUBLICO
# =============================================================================

# Iniciar sesion (devuelve accessToken + refreshToken)   [PUBLICO]
curl -X POST "$BASE/api/auth/login" -H "Content-Type: application/json" \
  -d '{"usuario":"admin","contrasena":"Admin@2026"}'

# Renovar el accessToken con el refreshToken   [PUBLICO]
curl -X POST "$BASE/api/auth/refresh" -H "Content-Type: application/json" \
  -d '{"refreshToken":"pega-el-refreshToken"}'

# Cerrar sesion (revoca el refreshToken)   [PUBLICO]
curl -X POST "$BASE/api/auth/logout" -H "Content-Type: application/json" \
  -d '{"refreshToken":"pega-el-refreshToken"}'


# =============================================================================
# ADMINISTRACION DE USUARIOS  (/api/admin)   ->   solo ADMIN
# =============================================================================

# Listar usuarios (paginado + filtros)   [ADMIN]
curl "$BASE/api/admin/usuarios?pagina=0&tamano=10&buscar=&rol=&estado=" -H "Authorization: Bearer $TOKEN"

# Obtener un usuario por id   [ADMIN]
curl "$BASE/api/admin/usuarios/1" -H "Authorization: Bearer $TOKEN"

# Crear usuario   [ADMIN]
curl -X POST "$BASE/api/admin/usuarios" -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"usuario":"jperez","nombre":"Juan","apellido":"Perez","correo":"jp@x.com","dni":"12345678","password":"Clave123","rolesIds":[2]}'

# Editar datos del usuario   [ADMIN]
curl -X PUT "$BASE/api/admin/usuarios/2" -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"nombre":"Juan","apellido":"Perez Lopez","correo":"jp@x.com","dni":"12345678"}'

# Eliminar usuario   [ADMIN]
curl -X DELETE "$BASE/api/admin/usuarios/2" -H "Authorization: Bearer $TOKEN"

# Activar / desactivar usuario   [ADMIN]
curl -X PATCH "$BASE/api/admin/usuarios/2/estado" -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"activo":false}'

# Asignar roles (reemplaza los actuales)   [ADMIN]
curl -X PATCH "$BASE/api/admin/usuarios/2/roles" -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"rolesIds":[2,4]}'

# Cambiar contrasena del usuario   [ADMIN]
curl -X PATCH "$BASE/api/admin/usuarios/2/password" -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"password":"NuevaClave123"}'

# Listar roles disponibles (para selects)   [ADMIN]
curl "$BASE/api/admin/roles" -H "Authorization: Bearer $TOKEN"


# =============================================================================
# RESULTADOS  (/api/resultados)   ->   ADMIN, RESULTADOS
# =============================================================================

# Cargar resultados (multipart). mensaje=texto (max 500). confirmarNombre=true fuerza si el PDF no coincide con RENIEC   [ADMIN, RESULTADOS]
curl -X POST "$BASE/api/resultados/upload" -H "Authorization: Bearer $TOKEN" \
  -F "phone=987654321" -F "dni=12345678" -F "patientName=Juan Perez" \
  -F "mensaje=Sus resultados estan listos" \
  -F "confirmarNombre=false" \
  -F "files=@/ruta/resultado.pdf;type=application/pdf"

# Consultar resultados (paginado + filtros DNI/nombre/fecha)   [ADMIN, RESULTADOS]
curl "$BASE/api/resultados?dni=&nombre=&desde=&hasta=&pagina=0&tamano=10" -H "Authorization: Bearer $TOKEN"

# Ver detalle de un resultado (con sus archivos)   [ADMIN, RESULTADOS]
curl "$BASE/api/resultados/1" -H "Authorization: Bearer $TOKEN"

# Actualizar resultado (telefono/nombre + reenviar SMS)   [ADMIN, RESULTADOS]
curl -X PUT "$BASE/api/resultados/1" -H "Authorization: Bearer $TOKEN" -H "Content-Type: application/json" \
  -d '{"telefono":"987654321","nombrePaciente":"Juan Perez","reenviarSms":true}'

# Consultar logs de carga (paginado + filtros DNI/fecha)   [ADMIN, RESULTADOS]
curl "$BASE/api/resultados/logs?dni=&desde=&hasta=&pagina=0&tamano=10" -H "Authorization: Bearer $TOKEN"


# =============================================================================
# PORTAL DEL PACIENTE  (/api/paciente)   ->   ADMIN, PACIENTE (solo lo suyo)
# =============================================================================

# Mis resultados (los del paciente autenticado, filtro por fecha)   [ADMIN, PACIENTE]
curl "$BASE/api/paciente/resultados?desde=&hasta=&pagina=0&tamano=10" -H "Authorization: Bearer $TOKEN"

# Detalle de un resultado propio   [ADMIN, PACIENTE]
curl "$BASE/api/paciente/resultados/1" -H "Authorization: Bearer $TOKEN"

# Ver un archivo propio en linea   [ADMIN, PACIENTE]
curl "$BASE/api/paciente/resultados/1/archivos/1" -H "Authorization: Bearer $TOKEN" -o archivo.pdf

# Descargar un archivo propio (registra la descarga)   [ADMIN, PACIENTE]
curl -OJ "$BASE/api/paciente/resultados/1/archivos/1/descargar" -H "Authorization: Bearer $TOKEN"


# =============================================================================
# ACCESO ANONIMO DEL PACIENTE (link del SMS)   ->   PUBLICO (secreto = token/code)
# =============================================================================

# Abrir link corto del SMS -> cuenta CLICK y redirige (302)   [PUBLICO]
curl -i "$BASE/s/{code}"

# Ver la pagina de resultados (HTML) -> marca visto   [PUBLICO]
curl "$BASE/r/{token}"

# Ver archivo inline por token   [PUBLICO]
curl "$BASE/r/{token}/archivo/1" -o archivo.pdf

# Descargar archivo por token   [PUBLICO]
curl -OJ "$BASE/r/{token}/descargar/1"


# =============================================================================
# DOCUMENTACION / SALUD   ->   PUBLICO
# =============================================================================

# Swagger UI (navegador):  $BASE/swagger-ui.html
# OpenAPI JSON   [PUBLICO]
curl "$BASE/v3/api-docs"

# Healthcheck   [PUBLICO]
curl "$BASE/actuator/health"
